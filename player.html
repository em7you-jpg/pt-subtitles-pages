#!/usr/bin/env python3 """ Perfect-sync generator (Whisper-1 timestamps → track.json) GOAL (대표님 요구사항 반영): - "버킷팅(resample) 기반" 싱크 깨짐을 제거한다. - 타임라인(start/end)은 Whisper word timestamp + 섹션 텍스트 매칭(align_sections)으로 만든다. - 표시 텍스트는 sections_payload의 원문 JA/KO를 그대로 사용한다. - "너무 빨리 사라짐", "두 줄이 한 번에 나오고 이후 엇나감", "초반에 몰림"을 구조적으로 줄인다. ⚠️ 이번 이슈의 확정 원인(대표님이 이미 jq로 증거 제공): - lines는 정상인데, cues가 "다음 라인 start(예: Intro placeholder 0.86)" 기준으로 잘려서 cues[0].end=0.81 같은 '초반 즉시 소멸'이 발생. - 즉, cues 생성에서 "빈 가사 placeholder(ja/ko 비어있음)"가 싱크를 망친다. FIX (반드시 반영): 1) cues는 '재계산(next_start 컷)'하지 않고, 이미 계산된 lines의 (start,end)를 그대로 복사해서 만든다. 2) cues 생성에서 ja/ko가 둘 다 빈 placeholder 라인은 제외한다. 3) player 호환을 위해 cues에 start/end 뿐 아니라 t0/t1도 함께 넣는다. 이 파일은 대표님이 붙여주신 "현재 최신 파일"을 기준으로 위 FIX를 정확히 반영한 통파일입니다. """ import argparse import json import os import re import sys from dataclasses import dataclass from datetime import datetime from pathlib import Path from typing import List, Optional, Tuple from openai import OpenAI # =============================== # Debug dump # =============================== DEBUG_DIR = Path("/opt/pt_whisper/site/debug") def dump_debug(name: str, payload): """ Write debug JSON safely. - Never raises. - If payload is not JSON-serializable, falls back to string representation. """ try: os.makedirs(DEBUG_DIR, exist_ok=True) ts = datetime.now().strftime("%Y%m%d_%H%M%S_%f") path = os.path.join(DEBUG_DIR, f"{ts}_{name}.json") # helpful for journalctl inspection print(f"[DEBUG_WRITE] {path}", flush=True) def _default(o): try: if hasattr(o, "model_dump"): return o.model_dump() except Exception: pass try: if hasattr(o, "dict"): return o.dict() except Exception: pass try: return str(o) except Exception: return "" with open(path, "w", encoding="utf-8") as f: json.dump(payload, f, ensure_ascii=False, indent=2, default=_default) except Exception: return # =============================== # Normalization helpers # =============================== def norm(s: str) -> str: # remove spaces and common punctuation; keep JP chars s = s.replace(" ", "") s = re.sub(r'[、。・「」『』（）()\[\]{}"“”’\'\n\r\t]', "", s) s = s.replace("／", "/") return s # =============================== # Data models # =============================== @dataclass class Word: w: str start: float end: float @dataclass class Segment: start: float end: float text: str # =============================== # Whisper transcription # =============================== def transcribe_words(client: OpenAI, audio_path: str, language: Optional[str]) -> List[Word]: """ Whisper word timestamps. """ with open(audio_path, "rb") as f: params = { "file": f, "model": "whisper-1", "response_format": "verbose_json", "timestamp_granularities": ["word"], } if language: params["language"] = language vt = client.audio.transcriptions.create(**params) # Debug: raw vt (best effort) try: raw_json = vt.model_dump_json() dump_debug("whisper_raw", {"params": params, "vt_raw_json": raw_json}) except Exception: try: dump_debug("whisper_raw", { "params": params, "has_words": bool(getattr(vt, "words", None)), "words_len": len(getattr(vt, "words", []) or []), "text_head": (getattr(vt, "text", "") or "")[:200], }) except Exception: pass words: List[Word] = [] for item in getattr(vt, "words", []) or []: w = getattr(item, "word", None) or (item.get("word") if isinstance(item, dict) else None) st = getattr(item, "start", None) or (item.get("start") if isinstance(item, dict) else None) en = getattr(item, "end", None) or (item.get("end") if isinstance(item, dict) else None) if st is None or en is None: continue try: st_f = float(st) en_f = float(en) except Exception: continue if en_f <= st_f: continue words.append(Word(str(w or ""), st_f, en_f)) if not words: raise RuntimeError( "No word timestamps returned. Ensure response_format=verbose_json and " "timestamp_granularities=['word']." ) # Debug: word summary try: head = [{"i": i, "w": words[i].w, "st": round(words[i].start, 3), "en": round(words[i].end, 3)} for i in range(min(60, len(words)))] dump_debug("words_head", {"count": len(words), "head": head}) except Exception: pass return words def transcribe_segments(client: OpenAI, audio_path: str, language: Optional[str]) -> List[Segment]: """ Kept only for debugging/inspection. Not used as primary alignment due to drift risk (버킷팅 원인). """ with open(audio_path, "rb") as f: params = { "file": f, "model": "whisper-1", "response_format": "verbose_json", "timestamp_granularities": ["word"], } if language: params["language"] = language vt = client.audio.transcriptions.create(**params) segs: List[Segment] = [] for s in getattr(vt, "segments", []) or []: st = getattr(s, "start", None) or (s.get("start") if isinstance(s, dict) else None) en = getattr(s, "end", None) or (s.get("end") if isinstance(s, dict) else None) tx = getattr(s, "text", None) or (s.get("text") if isinstance(s, dict) else "") if st is None or en is None: continue try: st_f = float(st) en_f = float(en) except Exception: continue if en_f <= st_f: continue segs.append(Segment(st_f, en_f, str(tx or ""))) if not segs: # fallback: words as "segments" for wobj in getattr(vt, "words", []) or []: w = getattr(wobj, "word", None) or (wobj.get("word") if isinstance(wobj, dict) else "") st = getattr(wobj, "start", None) or (wobj.get("start") if isinstance(wobj, dict) else None) en = getattr(wobj, "end", None) or (wobj.get("end") if isinstance(wobj, dict) else None) if st is None or en is None: continue try: st_f = float(st) en_f = float(en) except Exception: continue if en_f <= st_f: continue segs.append(Segment(st_f, en_f, str(w or ""))) if not segs: raise RuntimeError("No usable timestamps returned: both segments and words are empty.") try: seg_summary = [ {"i": i, "start": round(s.start, 3), "end": round(s.end, 3), "dur": round(s.end - s.start, 3), "text_head": (s.text or "")[:40]} for i, s in enumerate(segs[:60]) ] dump_debug("segments_summary", {"count": len(segs), "head": seg_summary}) except Exception: pass return segs # =============================== # Alignment (word-based) # =============================== def build_index(words: List[Word]) -> Tuple[str, List[int]]: """ Build concatenated string (no spaces/punct) and map char position -> word index. """ s = [] pos_to_word: List[int] = [] for i, wd in enumerate(words): token = norm(wd.w) if not token: continue for _ in token: pos_to_word.append(i) s.append(token) return "".join(s), pos_to_word def align_sections(words: List[Word], sections: List[dict]) -> Tuple[List[dict], List[str]]: """ Find each section.ja inside concatenated whisper words → section.start. This is the "precision" path (no resample bucket drift). """ concat, pos_to_word = build_index(words) cursor = 0 issues: List[str] = [] if not words: raise ValueError("No timed words from whisper result (all timestamps missing)") total_dur = max(w.end for w in words) out_lines: List[dict] = [] last_start = 0.0 for idx, sec in enumerate(sections): target = norm(sec.get("ja") or "") if not target: # empty lyric placeholder: keep it, but start will be filled later by end-builder out_lines.append({**sec, "start": last_start, "_match": "empty"}) continue found_at = concat.find(target, cursor) if found_at == -1: # estimated start (fallback) est = total_dur * (idx / max(1, len(sections))) issues.append( f"[WARN] Section {idx} not matched; using estimated start {est:.2f}s: " f"{sec.get('tag')} {(sec.get('ja') or '')[:24]}..." ) out_lines.append({**sec, "start": round(est, 3), "_match": "estimated"}) continue start_char = found_at end_char = found_at + len(target) - 1 w_start_i = pos_to_word[start_char] start_t = words[w_start_i].start out_lines.append({**sec, "start": round(start_t, 3), "_match": "ok"}) last_start = start_t cursor = end_char + 1 # monotonic clamp prev = -1.0 for l in out_lines: st = float(l.get("start") or 0.0) if st < prev: l["start"] = round(prev, 3) l["_match"] = "clamped" prev = float(l["start"]) return out_lines, issues # =============================== # Read-time end / cue builder (stable) # =============================== def _has_text_line(ln: dict) -> bool: ja = (ln.get("ja") or "").strip() ko = (ln.get("ko") or "").strip() txt = (ln.get("text") or "").strip() return bool(ja or ko or txt) def build_end_and_merge_lines( lines: List[dict], issues: List[str], *, lead_start_sec: float = 0.7, sec_per_char: float = 0.065, min_show_sec: float = 2.2, max_show_sec: float = 18.0, full_gap_sec: float = 8.0, max_last_sec: float = 18.0, pad_end: float = 0.12, ) -> List[dict]: """ - Apply small global lead on START only (avoid ending too early). - Compute END using next START and read-time. - If next start is too soon (causes 0.x sec cues), merge adjacent lines to avoid flicker. IMPORTANT: - Empty placeholders (ja/ko empty) are kept in lines output (for HUD/ordering), BUT they must NOT influence cues shortening. (That fix is in build_cues_from_lines) """ # Apply lead to starts (non-empty lyrics only) for l in lines: ja_txt = str(l.get("ja") or "").strip() if not ja_txt: continue st = float(l.get("start") or 0.0) - float(lead_start_sec) if st < 0.0: st = 0.0 l["start"] = round(st, 3) # Sort by time tmp: List[Tuple[float, dict]] = [] for l in lines: try: st = float(l.get("start") or 0.0) except Exception: st = 0.0 tmp.append((st, l)) tmp.sort(key=lambda x: x[0]) merged_out: List[dict] = [] i = 0 while i < len(tmp): st = tmp[i][0] l = dict(tmp[i][1]) # copy # find next start next_st = tmp[i + 1][0] if (i + 1) < len(tmp) else None ja = (l.get("ja") or "").strip() ko = (l.get("ko") or "").strip() txt_src = ja or ko or (l.get("text") or "").strip() char_len = len(txt_src) show_sec = max(min_show_sec, min(char_len * sec_per_char, max_show_sec)) # merge condition: next comes too soon -> current would be unreadable # merge only if both lines have any text; keep empties as separators if next_st is not None: gap = next_st - st if gap < (min_show_sec + 0.25): l2 = dict(tmp[i + 1][1]) if _has_text_line(l) and _has_text_line(l2): ja2 = (l2.get("ja") or "").strip() ko2 = (l2.get("ko") or "").strip() ja_merged = "\n".join([x for x in [ja, ja2] if x]) ko_merged = "\n".join([x for x in [ko, ko2] if x]) l["ja"] = ja_merged l["ko"] = ko_merged l["_match"] = "merged_short_gap" # use next2 for end next2 = tmp[i + 2][0] if (i + 2) < len(tmp) else None if next2 is None: ed = st + min(show_sec, max_last_sec) else: gap2 = next2 - st if gap2 <= full_gap_sec: ed = max(st, next2 - pad_end) else: ed = min(st + show_sec, max(st, next2 - pad_end)) l["start"] = round(st, 3) l["end"] = round(ed, 3) merged_out.append(l) i += 2 continue # normal end if next_st is None: ed = st + min(show_sec, max_last_sec) else: gap = next_st - st if gap <= full_gap_sec: ed = max(st, next_st - pad_end) else: ed = min(st + show_sec, max(st, next_st - pad_end)) l["start"] = round(st, 3) l["end"] = round(ed, 3) if "_match" not in l: l["_match"] = "ok_words" merged_out.append(l) i += 1 issues.append( f"[INFO] WORD_ALIGN lead_start={lead_start_sec:.2f}s sec_per_char={sec_per_char:.3f} " f"min_show={min_show_sec:.2f} full_gap={full_gap_sec:.2f} pad_end={pad_end:.2f}" ) # Debug: duration stats (non-empty only) try: durs = [] for l in merged_out: ja_txt = (l.get("ja") or "").strip() if not ja_txt: continue d = float(l.get("end") or 0.0) - float(l.get("start") or 0.0) if d > 0: durs.append(d) if durs: issues.append(f"[INFO] DUR_STATS_NONEMPTY min={min(durs):.2f} avg={(sum(durs)/len(durs)):.2f} max={max(durs):.2f}") except Exception: pass return merged_out def build_cues_from_lines(lines: List[dict], issues: List[str]) -> List[dict]: """ Player가 cues를 우선 사용할 수 있으므로, lines로부터 "정확한" cues 생성. ✅ FIX V3 (대표님 제공 증거 기반): - cues의 end를 'next_start'로 다시 계산하지 않는다. (그게 Intro placeholder start=0.86 때문에 cues[0].end=0.81로 잘리는 원인이었음) - 대신 lines에서 이미 계산된 (start,end)를 그대로 복사한다. - ja/ko 둘 다 빈 placeholder 라인은 cues에서 제외한다. - player 호환을 위해 start/end 뿐 아니라 t0/t1도 동시에 제공한다. """ cues: List[dict] = [] for l in lines: if not isinstance(l, dict): continue ja = (l.get("ja") or "").strip() ko = (l.get("ko") or "").strip() # ✅ placeholder 제거: 빈 Intro - instrumental 등 if not (ja or ko): continue try: st = float(l.get("start") or 0.0) ed = float(l.get("end") or 0.0) except Exception: continue # 방어: 혹시 end가 없거나 start==end면 최소 표시시간 보장 if ed <= st: ed = st + 2.2 if ja and ko: txt = ja + "\n" + ko else: txt = ja or ko cues.append({ # ✅ dual schema "t0": round(st, 2), "t1": round(ed, 2), "start": round(st, 2), "end": round(ed, 2), "text": txt, "ja": ja, "ko": ko, }) issues.append(f"[INFO] CUES_BUILT={len(cues)} (from lines exact, placeholders skipped)") # debug head try: dump_debug("cues_head", {"cues_len": len(cues), "head": cues[:10]}) except Exception: pass return cues # =============================== # CLI main # =============================== def main(): ap = argparse.ArgumentParser() ap.add_argument("--audio", required=True, help="mp3/wav path") ap.add_argument("--sections", required=True, help="lyrics_sections.json") ap.add_argument("--out", default="track.json", help="output track json") ap.add_argument("--language", default="ja", help="language hint (e.g., ja)") ap.add_argument("--title", default=None, help="override title in output") args = ap.parse_args() if not os.getenv("OPENAI_API_KEY"): print("ERROR: OPENAI_API_KEY env var not set.", file=sys.stderr) sys.exit(2) client = OpenAI() with open(args.sections, "r", encoding="utf-8") as f: data = json.load(f) sections = data.get("sections") or [] title = args.title or data.get("title") or "(set title later)" words = transcribe_words(client, args.audio, args.language) raw_lines, issues = align_sections(words, sections) # end + merge + cues tuned_lines = build_end_and_merge_lines(raw_lines, issues) cues = build_cues_from_lines(tuned_lines, issues) out = { "title": title, "lines": [ { "start": l.get("start"), "end": l.get("end"), "ja": l.get("ja", ""), "ko_pron": l.get("ko_pron", ""), "ko": l.get("ko", ""), "tag": l.get("tag"), "_match": l.get("_match", ""), } for l in tuned_lines ], "cues": cues, "meta": { "engine": "word_align_v3_cues_exact", "generated_at": datetime.utcnow().isoformat() + "Z", } } # Debug snapshots try: dump_debug("final_lines", {"title": title, "lines_len": len(out["lines"]), "head": out["lines"][:40]}) dump_debug("final_cues", {"cues_len": len(out["cues"]), "head": out["cues"][:20]}) except Exception: pass with open(args.out, "w", encoding="utf-8") as f: json.dump(out, f, ensure_ascii=False, indent=2) print(f"OK: wrote {args.out} with {len(out['lines'])} lines, {len(out.get('cues', []))} cues.") if issues: print("\n".join(issues)) # =============================== # API helper (importable function) # =============================== def generate_track_json(audio_path: str, sections_payload: dict, language: str = "ja", title_override: str = None): """ audio_path: local mp3/wav path on server sections_payload: dict like: { "title": "...", "sections": [ {"tag":"Verse 1","ja":"...", "ko":"", "ko_pron":""}, ... ] } returns: (out, issues_list) """ if not os.getenv("OPENAI_API_KEY"): raise RuntimeError("OPENAI_API_KEY env var not set.") client = OpenAI() sections = sections_payload.get("sections") or [] if not isinstance(sections, list) or not sections: raise RuntimeError("sections_payload.sections is missing or empty.") title = title_override or sections_payload.get("title") or "(set title later)" # Primary path: WORD alignment (precision) words = transcribe_words(client, audio_path, language) raw_lines, issues = align_sections(words, sections) tuned_lines = build_end_and_merge_lines(raw_lines, issues) cues = build_cues_from_lines(tuned_lines, issues) out = { "title": title, "lines": [ { "start": l.get("start"), "end": l.get("end"), "ja": l.get("ja", ""), "ko_pron": l.get("ko_pron", ""), "ko": l.get("ko", ""), "tag": l.get("tag"), "_match": l.get("_match", ""), } for l in tuned_lines ], "cues": cues, "meta": { "engine": "word_align_v3_cues_exact", "generated_at": datetime.utcnow().isoformat() + "Z", } } # Debug snapshots try: dump_debug("api_final_lines", {"title": title, "lines_len": len(out["lines"]), "head": out["lines"][:40]}) dump_debug("api_final_cues", {"cues_len": len(out["cues"]), "head": out["cues"][:20]}) except Exception: pass return out, issues if __name__ == "__main__": main()
